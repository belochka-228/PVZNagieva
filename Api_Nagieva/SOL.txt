CREATE TABLE Roles (

    RoleName NVARCHAR(50) NOT NULL UNIQUE

GO

CREATE TABLE OrderStatuses (
    StatusId   INT IDENTITY(1,1) PRIMARY KEY,
    StatusName NVARCHAR(50) NOT NULL UNIQUE
);
GO

CREATE TABLE DeliveryTypes (
    DeliveryTypeId   INT IDENTITY(1,1) PRIMARY KEY,
    DeliveryTypeName NVARCHAR(50) NOT NULL UNIQUE
);
GO

CREATE TABLE Categories (
    CategoryId   INT IDENTITY(1,1) PRIMARY KEY,
    CategoryName NVARCHAR(100) NOT NULL UNIQUE,
    Description  NVARCHAR(500) NULL
);
GO

CREATE TABLE PickupPoints (
    PickupPointId INT IDENTITY(1,1) PRIMARY KEY,
    City          NVARCHAR(100) NOT NULL,
    Address       NVARCHAR(300) NOT NULL,
    Phone         NVARCHAR(30)  NULL,
    WorkingHours  NVARCHAR(100) NULL,
    IsActive      BIT NOT NULL DEFAULT 1
);
GO

-- ============================================================
--  ПОЛЬЗОВАТЕЛИ
-- ============================================================

CREATE TABLE Users (
    UserId       INT IDENTITY(1,1) PRIMARY KEY,
    Login        NVARCHAR(100) NOT NULL UNIQUE,
    PasswordHash NVARCHAR(255) NOT NULL,
    RoleId       INT NOT NULL,
    CreatedAt    DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT FK_Users_Roles FOREIGN KEY (RoleId) REFERENCES Roles(RoleId)
);
GO

-- ============================================================
--  ТОВАРЫ
-- ============================================================

CREATE TABLE Products (
    ProductId        INT IDENTITY(1,1) PRIMARY KEY,
    Name             NVARCHAR(200)  NOT NULL,
    Description      NVARCHAR(2000) NULL,
    Price            DECIMAL(18,2)  NOT NULL,
    CategoryId       INT            NOT NULL,
    ImagePath        NVARCHAR(300)  NULL,
    StockQuantity    INT            NOT NULL DEFAULT 0,
    UnitsPerPackage  INT            NULL,
    IsAvailable      BIT            NOT NULL DEFAULT 1,
    Rating           DECIMAL(3,2)   NOT NULL DEFAULT 0,
    CreatedAt        DATETIME2      NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT FK_Products_Categories FOREIGN KEY (CategoryId) REFERENCES Categories(CategoryId),
    CONSTRAINT CK_Products_Price      CHECK (Price >= 0),
    CONSTRAINT CK_Products_Stock      CHECK (StockQuantity >= 0)
);
GO

CREATE INDEX IX_Products_Name     ON Products(Name);
CREATE INDEX IX_Products_Category ON Products(CategoryId);
CREATE INDEX IX_Products_Price    ON Products(Price);
GO

-- ============================================================
--  ОТЗЫВЫ
-- ============================================================

CREATE TABLE Reviews (
    ReviewId  INT IDENTITY(1,1) PRIMARY KEY,
    ProductId INT NOT NULL,
    UserId    INT NOT NULL,
    Rating    INT NOT NULL,
    Comment   NVARCHAR(2000) NULL,
    CreatedAt DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT FK_Reviews_Products FOREIGN KEY (ProductId) REFERENCES Products(ProductId) ON DELETE CASCADE,
    CONSTRAINT FK_Reviews_Users    FOREIGN KEY (UserId)    REFERENCES Users(UserId),
    CONSTRAINT CK_Reviews_Rating   CHECK (Rating BETWEEN 1 AND 5),
    CONSTRAINT UQ_Reviews_UserProduct UNIQUE (ProductId, UserId)
);
GO

-- Триггер для пересчета среднего рейтинга
CREATE TRIGGER trg_Reviews_UpdateRating
ON Reviews
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    UPDATE p
    SET p.Rating = ISNULL((
        SELECT AVG(CAST(r.Rating AS DECIMAL(3,2)))
        FROM Reviews r WHERE r.ProductId = p.ProductId
    ), 0)
    FROM Products p
    WHERE p.ProductId IN (SELECT ProductId FROM inserted
                          UNION
                          SELECT ProductId FROM deleted);
END;
GO

-- ============================================================
--  ИЗБРАННОЕ И КОРЗИНА
-- ============================================================

CREATE TABLE Favorites (
    FavoriteId INT IDENTITY(1,1) PRIMARY KEY,
    UserId     INT NOT NULL,
    ProductId  INT NOT NULL,
    AddedAt    DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT FK_Fav_Users    FOREIGN KEY (UserId)    REFERENCES Users(UserId)    ON DELETE CASCADE,
    CONSTRAINT FK_Fav_Products FOREIGN KEY (ProductId) REFERENCES Products(ProductId) ON DELETE CASCADE,
    CONSTRAINT UQ_Favorites UNIQUE (UserId, ProductId)
);
GO

CREATE TABLE CartItems (
    CartItemId INT IDENTITY(1,1) PRIMARY KEY,
    UserId     INT NOT NULL,
    ProductId  INT NOT NULL,
    Quantity   INT NOT NULL DEFAULT 1,
    AddedAt    DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    CONSTRAINT FK_Cart_Users    FOREIGN KEY (UserId)    REFERENCES Users(UserId)    ON DELETE CASCADE,
    CONSTRAINT FK_Cart_Products FOREIGN KEY (ProductId) REFERENCES Products(ProductId) ON DELETE CASCADE,
    CONSTRAINT UQ_Cart UNIQUE (UserId, ProductId),
    CONSTRAINT CK_Cart_Quantity CHECK (Quantity > 0)
);
GO

-- ============================================================
--  ЗАКАЗЫ
-- ============================================================

CREATE TABLE Orders (
    OrderId         INT IDENTITY(1,1) PRIMARY KEY,
    UserId          INT NOT NULL,
    OrderDate       DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    TotalAmount     DECIMAL(18,2) NOT NULL,
    DeliveryTypeId  INT NOT NULL,
    DeliveryAddress NVARCHAR(500) NULL,
    PickupPointId   INT NULL,
    StatusId        INT NOT NULL,
    CONSTRAINT FK_Orders_Users        FOREIGN KEY (UserId)         REFERENCES Users(UserId),
    CONSTRAINT FK_Orders_DelTypes     FOREIGN KEY (DeliveryTypeId) REFERENCES DeliveryTypes(DeliveryTypeId),
    CONSTRAINT FK_Orders_PickupPoints FOREIGN KEY (PickupPointId)  REFERENCES PickupPoints(PickupPointId),
    CONSTRAINT FK_Orders_Statuses     FOREIGN KEY (StatusId)       REFERENCES OrderStatuses(StatusId),
    CONSTRAINT CK_Orders_Total        CHECK (TotalAmount >= 0),
    CONSTRAINT CK_Orders_MinDelivery  CHECK (TotalAmount >= 600 OR DeliveryTypeId = 2),
    CONSTRAINT CK_Orders_DeliveryInfo CHECK (
        (DeliveryTypeId = 1 AND DeliveryAddress IS NOT NULL) OR
        (DeliveryTypeId = 2 AND PickupPointId IS NOT NULL)
    )
);
GO

CREATE TABLE OrderItems (
    OrderItemId     INT IDENTITY(1,1) PRIMARY KEY,
    OrderId         INT NOT NULL,
    ProductId       INT NOT NULL,
    ProductName     NVARCHAR(200) NOT NULL,
    Quantity        INT NOT NULL,
    PriceAtPurchase DECIMAL(18,2) NOT NULL,
    CONSTRAINT FK_OI_Orders   FOREIGN KEY (OrderId)   REFERENCES Orders(OrderId)   ON DELETE CASCADE,
    CONSTRAINT FK_OI_Products FOREIGN KEY (ProductId) REFERENCES Products(ProductId),
    CONSTRAINT CK_OI_Quantity CHECK (Quantity > 0),
    CONSTRAINT CK_OI_Price    CHECK (PriceAtPurchase >= 0)
);
GO

-- ============================================================
--  ПРЕДСТАВЛЕНИЕ (Исправлено: теперь идет отдельным пакетом)
-- ============================================================

CREATE VIEW vw_ProductDetails AS
SELECT  p.ProductId, p.Name, p.Description, p.Price,
        c.CategoryName, p.ImagePath, p.StockQuantity,
        p.UnitsPerPackage, p.IsAvailable, p.Rating,
        (SELECT COUNT(*) FROM Reviews r WHERE r.ProductId = p.ProductId) AS ReviewCount
FROM Products p
JOIN Categories c ON c.CategoryId = p.CategoryId;
GO
